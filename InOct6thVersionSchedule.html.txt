            scheduleData.forEach(d => {
                const isActive = selectedDayNum === d.dayNum;
                html += `
                    <button onclick="selectDay(${d.dayNum})" class="px-4 py-2 rounded-xl text-xs font-bold whitespace-nowrap transition ${isActive ? 'bg-rose-600 text-white shadow-md shadow-rose-200' : 'bg-white text-slate-600 border border-slate-200 hover:bg-slate-100'}">

                    </button>
                `;
            });

            container.innerHTML = html;
        }

        // Select Day Filter
        function selectDay(dayNum) {
            selectedDayNum = dayNum;
            renderDayTabs();
            filterSchedule();
        }

        // Filter and Render Schedule Events
        function filterSchedule() {
            const searchVal = document.getElementById('scheduleSearch').value.toLowerCase();
            const categoryVal = document.getElementById('categoryFilter').value;
            const container = document.getElementById('scheduleCardsContainer');

            let filteredDays = scheduleData;
            if (selectedDayNum !== -1) {
                filteredDays = scheduleData.filter(d => d.dayNum === selectedDayNum);
            }

            let cardsHtml = '';
            let totalMatchCount = 0;

            filteredDays.forEach(day => {
                const matchingEvents = day.events.filter(ev => {
                    const matchSearch = ev.titleEn.toLowerCase().includes(searchVal) ||
                                        ev.titleJp.toLowerCase().includes(searchVal) ||
                                        ev.venue.toLowerCase().includes(searchVal) ||
                                        ev.coordinator.toLowerCase().includes(searchVal);
                    const matchCategory = categoryVal === 'ALL' || ev.category === categoryVal;
                    return matchSearch && matchCategory;
                });

                if (matchingEvents.length > 0) {
                    totalMatchCount += matchingEvents.length;
                    cardsHtml += `
                        <div class="bg-white rounded-3xl border border-slate-200/90 p-5 sm:p-6 shadow-sm space-y-4">
                            <div class="flex flex-wrap items-center justify-between gap-2 border-b border-slate-100 pb-3">
                                <div class="flex items-center gap-3">
                                    <span class="bg-rose-100 text-rose-800 text-xs font-black px-3 py-1 rounded-xl uppercase">Day ${day.dayNum}</span>
                                    <h3 class="text-base sm:text-lg font-bold text-slate-900">${currentLang === 'en' ? day.date : day.dateJp}</h3>
                                </div>
                                <span class="text-xs text-slate-400 font-medium">${matchingEvents.length} ${currentLang === 'en' ? 'Activities' : '項目'}</span>
                            </div>

                            <div class="divide-y divide-slate-100">
                                ${matchingEvents.map(ev => getEventRowHtml(ev)).join('')}
                            </div>
                        </div>
                    `;
                }
            });

            if (totalMatchCount === 0) {
                container.innerHTML = `
                    <div class="bg-white rounded-3xl border border-dashed border-slate-300 p-12 text-center text-slate-400 space-y-2">
                        <i class="fa-solid fa-calendar-xmark text-4xl text-slate-300"></i>
                        <p class="text-sm font-semibold">${currentLang === 'en' ? 'No activities match your filter.' : '該当するスケジュールは見つかりませんでした。'}</p>
                    </div>
                `;
            } else {
                container.innerHTML = cardsHtml;
            }
        }

        // Return Item Row Badge Color
        function getCategoryBadge(cat) {
            switch (cat) {
                case 'Culture': return 'bg-rose-50 text-rose-700 border-rose-200';
                case 'Excursion': return 'bg-amber-50 text-amber-700 border-amber-200';
                case 'Sports': return 'bg-emerald-50 text-emerald-700 border-emerald-200';
                case 'Academic': return 'bg-indigo-50 text-indigo-700 border-indigo-200';
                case 'Host': return 'bg-teal-50 text-teal-700 border-teal-200';
                default: return 'bg-slate-50 text-slate-700 border-slate-200';
            }
        }

        // Event Row Markup Generator
        function getEventRowHtml(ev) {
            const badgeStyle = getCategoryBadge(ev.category);
            const title = currentLang === 'en' ? ev.titleEn : ev.titleJp;
            const notes = currentLang === 'en' ? ev.notesEn : ev.notesJp;

            return `
                <div class="py-3.5 first:pt-0 last:pb-0 flex flex-col sm:flex-row sm:items-start justify-between gap-3 hover:bg-slate-50/80 p-2 rounded-2xl transition">
                    <div class="flex items-start gap-3 sm:w-1/4 shrink-0">
                        <span class="inline-block bg-slate-900 text-white text-xs font-mono font-bold px-2.5 py-1 rounded-lg">
                            ${ev.time}
                        </span>
                        <span class="text-[11px] font-bold px-2.5 py-0.5 rounded-full border ${badgeStyle} sm:hidden">
                            ${ev.category}
                        </span>
                    </div>

                    <div class="sm:w-2/4 space-y-1">
                        <h4 class="text-sm font-bold text-slate-800 leading-snug">${title}</h4>
                        <div class="flex flex-wrap items-center gap-3 text-xs text-slate-500">
                            <span><i class="fa-solid fa-location-dot text-rose-500 mr-1"></i>${ev.venue}</span>
                            <span><i class="fa-solid fa-user-check text-indigo-500 mr-1"></i>${ev.coordinator}</span>
                        </div>
                        ${notes ? `<p class="text-xs text-slate-500 bg-slate-100/80 px-2.5 py-1 rounded-lg italic mt-1">${notes}</p>` : ''}
                    </div>

                    <div class="hidden sm:flex sm:w-1/4 justify-end">
                        <span class="text-xs font-bold px-2.5 py-1 rounded-full border ${badgeStyle} h-fit">
                            ${ev.category}
                        </span>
                    </div>
                </div>
            `;
        }

        // Toggle Language Function
        function toggleLanguage() {
            currentLang = currentLang === 'en' ? 'jp' : 'en';
            document.getElementById('langLabel').innerText = currentLang === 'en' ? '日本語に切り替え' : 'Switch to English';

            // Update static elements with data-en / data-jp attributes
            document.querySelectorAll('[data-en]').forEach(el => {
                if (currentLang === 'en') {
                    el.innerText = el.getAttribute('data-en');
                } else {
                    el.innerText = el.getAttribute('data-jp');
                }
            });

            // Re-render components
            renderCoordinators();
            renderDayTabs();
            filterSchedule();
        }

        function updateCountdown() {
            const eventDate = new Date('October 08, 2026 16:20:00 GMT+0530').getTime();
            const now = new Date().getTime();
            const diff = eventDate - now;

            if (diff <= 0) {
                document.getElementById('cd-days').innerText = '00';
                document.getElementById('cd-hours').innerText = '00';
                document.getElementById('cd-mins').innerText = '00';
                document.getElementById('cd-secs').innerText = '00';
                return;
            }

            const days = Math.floor(diff / (1000 * 60 * 60 * 24));
            const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
            const secs = Math.floor((diff % (1000 * 60)) / 1000);

            document.getElementById('cd-days').innerText = String(days).padStart(2, '0');
            document.getElementById('cd-hours').innerText = String(hours).padStart(2, '0');
            document.getElementById('cd-mins').innerText = String(mins).padStart(2, '0');
            document.getElementById('cd-secs').innerText = String(secs).padStart(2, '0');
        }

        // Initialization
        window.onload = function() {
            renderCoordinators();
            renderDayTabs();
            filterSchedule();
            updateCountdown();
            setInterval(updateCountdown, 1000);
        };
    </script>
