    // Render Day Navigation Pills
    function renderDayPills() {
      const container = document.getElementById('day-pills-container');
      const texts = i18n[currentLang];
      
      let html = `
        <button onclick="selectDayFilter('all')" class="px-4 py-2 text-xs font-semibold rounded-xl whitespace-nowrap transition-all ${selectedDayFilter === 'all' ? 'bg-brand-600 text-white shadow-md' : 'bg-white text-slate-700 hover:bg-slate-100 border border-slate-200'}">
          ${texts.allDays}
        </button>
      `;

      itineraryData.forEach(day => {
        const isSelected = selectedDayFilter === day.dayNum;
        const dayLabel = currentLang === 'en' ? `Day ${day.dayNum}` : `${day.dayNum}日目`;
        html += `
          <button onclick="selectDayFilter(${day.dayNum})" class="px-3.5 py-2 text-xs font-medium rounded-xl whitespace-nowrap transition-all ${isSelected ? 'bg-brand-600 text-white shadow-md' : 'bg-white text-slate-600 hover:bg-slate-100 border border-slate-200'}">
            ${dayLabel}
          </button>
        `;
      });

      container.innerHTML = html;
    }

    // Filter by Day Pill
    function selectDayFilter(dayNum) {
      selectedDayFilter = dayNum;
      renderDayPills();
      renderItinerary();
    }

    // Quick Tag Filters
    function filterByTag(tagName) {
      selectedTagFilter = tagName;
      document.querySelectorAll('.tag-btn').forEach(btn => {
        if (btn.getAttribute('data-tag') === tagName) {
          btn.classList.add('bg-brand-600', 'text-white', 'border-brand-600');
          btn.classList.remove('bg-white', 'text-slate-700');
        } else {
          btn.classList.remove('bg-brand-600', 'text-white', 'border-brand-600');
          btn.classList.add('bg-white', 'text-slate-700');
        }
      });
      renderItinerary();
    }

    // Search Logic
    function handleSearch() {
      renderItinerary();
    }

    // Render Itinerary Cards
    function renderItinerary() {
      const container = document.getElementById('itinerary-list');
      const query = document.getElementById('search-input').value.toLowerCase().trim();
      const texts = i18n[currentLang];

      let filteredData = itineraryData;

      // Filter by Day
      if (selectedDayFilter !== 'all') {
        filteredData = filteredData.filter(d => d.dayNum === selectedDayFilter);
      }

      // Filter by Tag
      if (selectedTagFilter !== 'all') {
        filteredData = filteredData.filter(d => d.tags && d.tags.includes(selectedTagFilter));
      }

      let cardsHtml = '';
      let matchCount = 0;

      filteredData.forEach(day => {
        // Filter events by query if search active
        let matchingEvents = day.events;
        if (query) {
          matchingEvents = day.events.filter(ev => {
            const title = (currentLang === 'en' ? ev.titleEn : ev.titleJa) || '';
            const venue = (currentLang === 'en' ? ev.venueEn : ev.venueJa) || '';
            const coord = (currentLang === 'en' ? ev.coordinatorsEn : ev.coordinatorsJa) || '';
            const peon = (currentLang === 'en' ? ev.peonsEn : ev.peonsJa) || '';
            return title.toLowerCase().includes(query) || 
                   venue.toLowerCase().includes(query) || 
                   coord.toLowerCase().includes(query) || 
                   peon.toLowerCase().includes(query);
          });
        }

        if (matchingEvents.length === 0) return;
        matchCount++;

        const dayTitle = currentLang === 'en' ? day.titleEn : day.titleJa;
        const dayDate = currentLang === 'en' ? day.date : day.dateJa;
        const dayBadge = currentLang === 'en' ? `DAY ${day.dayNum}` : `${day.dayNum}日目`;

        cardsHtml += `
          <!-- Day Card -->
          <div class="bg-white rounded-2xl border border-slate-200/90 shadow-sm overflow-hidden transition-all duration-200 hover:shadow-md">
            
            <!-- Card Day Header -->
            <div class="bg-slate-900 text-white p-4 sm:p-5 flex flex-wrap items-center justify-between gap-3">
              <div class="flex items-center space-x-3">
                <span class="bg-brand-600 text-white text-xs font-black px-2.5 py-1 rounded-lg tracking-wider">
                  ${dayBadge}
                </span>
                <h3 class="text-base sm:text-lg font-bold">
                  ${dayTitle}
                </h3>
              </div>
              <div class="text-xs text-slate-300 font-medium flex items-center">
                <i class="fa-regular fa-calendar-check mr-1.5 text-brand-400"></i> ${dayDate}
              </div>
            </div>

            <!-- Card Events Body -->
            <div class="divide-y divide-slate-100 p-2 sm:p-4">
              ${matchingEvents.map(ev => {
                const title = currentLang === 'en' ? ev.titleEn : ev.titleJa;
                const venue = currentLang === 'en' ? ev.venueEn : ev.venueJa;
                const coord = currentLang === 'en' ? ev.coordinatorsEn : ev.coordinatorsJa;
                const peon = currentLang === 'en' ? ev.peonsEn : ev.peonsJa;
                const vehicle = currentLang === 'en' ? ev.vehicleEn : ev.vehicleJa;
                const note = currentLang === 'en' ? ev.noteEn : ev.noteJa;
                const stay = currentLang === 'en' ? ev.stayEn : ev.stayJa;

                return `
                  <div class="py-3 sm:py-4 px-2 sm:px-3 hover:bg-slate-50/80 rounded-xl transition-colors">
                    <div class="flex flex-col sm:flex-row sm:items-start gap-2 sm:gap-4">
                      
                      <!-- Time Pillar -->
                      <div class="sm:w-32 shrink-0">
                        <span class="inline-flex items-center text-xs font-bold text-brand-700 bg-brand-50 border border-brand-100 px-2 py-0.5 rounded-md">
                          <i class="fa-regular fa-clock mr-1 text-[10px]"></i> ${ev.time}
                        </span>
                      </div>

                      <!-- Content Column -->
                      <div class="flex-grow space-y-1.5">
                        <h4 class="text-sm font-semibold text-slate-900 leading-snug">
                          ${title}
                        </h4>

                        <!-- Badges and Details -->
                        <div class="flex flex-wrap gap-y-1.5 gap-x-4 text-xs text-slate-600 pt-0.5">
                          ${venue ? `
                            <div class="flex items-center text-slate-600">
                              <i class="fa-solid fa-location-dot text-rose-500 mr-1"></i>
                              <span class="font-medium">${venue}</span>
                            </div>
                          ` : ''}

                          ${vehicle ? `
                            <div class="flex items-center text-indigo-600">
                              <i class="fa-solid fa-van-shuttle mr-1"></i>
                              <span>${vehicle}</span>
                            </div>
                          ` : ''}

                          ${coord ? `
                            <div class="flex items-center text-slate-600">
                              <i class="fa-solid fa-user-check text-emerald-600 mr-1"></i>
                              <span><strong class="text-slate-500">${texts.coordinators}:</strong> ${coord}</span>
                            </div>
                          ` : ''}

                          ${peon ? `
                            <div class="flex items-center text-slate-500">
                              <i class="fa-solid fa-handshake-angle text-amber-600 mr-1"></i>
                              <span><strong class="text-slate-500">${texts.peons}:</strong> ${peon}</span>
                            </div>
                          ` : ''}
                        </div>

                        ${stay ? `
                          <div class="inline-block mt-1 text-[11px] font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 px-2 py-0.5 rounded-md">
                            <i class="fa-solid fa-house-user mr-1"></i> ${stay}
                          </div>
                        ` : ''}

                        ${note ? `
                          <div class="mt-1 text-xs text-amber-800 bg-amber-50/70 border border-amber-200/60 p-2 rounded-lg flex items-start space-x-1.5">
                            <i class="fa-solid fa-note-sticky text-amber-600 text-xs mt-0.5"></i>
                            <span>${note}</span>
                          </div>
                        ` : ''}
                      </div>
                    </div>
                  </div>
                `;
              }).join('')}
            </div>
          </div>
        `;
      });

      if (matchCount === 0) {
        container.innerHTML = `
          <div class="bg-white rounded-2xl border border-slate-200 p-8 text-center text-slate-500">
            <i class="fa-solid fa-calendar-xmark text-4xl text-slate-300 mb-3"></i>
            <p class="text-sm font-medium">${texts.noResults}</p>
          </div>
        `;
      } else {
        container.innerHTML = cardsHtml;
      }
    }

    // Initial Load
    window.onload = function() {
      updateUIStrings();
      renderDayPills();
      renderItinerary();
    };
  </script>
